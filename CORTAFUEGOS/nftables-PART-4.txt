#!/usr/sbin/nft -f

# =============================
# PARTE 4: EXPOSICION.
#
# Esta es la unica parte que abre los senuelos a Internet. Hasta aqui el
# forward tenia policy drop y solo reglas de EGRESS, asi que el trafico que
# llegaba por DNAT hacia los contenedores moria en esa cadena: eso era lo que
# mantenia todo sellado.
#
# Lo que cambia respecto a la parte 3:
#
#   1. Se anade un bloque de INGRESS en forward: los cinco puertos de los dos
#      senuelos (2222 y 2223 de Cowrie, 135/445/1433 de Dionaea), y nada mas.
#   2. Se prohibe explicitamente el trafico de contenedor a contenedor.
#   3. Se anade la segunda IP publica al anti pivoteo, que solo cubria la
#      primera.
#   4. Se quita el "tcp dport 22 accept" de input, que no hacia nada.
#
# Los puertos del INGRESS son los del CONTENEDOR, no los publicados: cuando el
# paquete llega a forward, el DNAT de la tabla nat ya ha reescrito el destino.
# Por eso aqui se ve 2222 en lugar de 22, y 2223 en lugar de 23.
#
# El 2223 es telnet. Tiene que estar aqui Y publicado en el compose: son las
# dos mitades de la misma exposicion, y si falta cualquiera de las dos el
# puerto sale "filtered" en un nmap. Todo el trabajo del sistema de ficheros
# ---el banner de issue.net, la linea de inetd.conf, el servidor en
# /usr/sbin--- no se ve desde fuera si el paquete no llega.
# =============================

add table inet filter
delete table inet filter

table inet filter {
    chain input {
        type filter hook input priority 0; policy drop;
        ct state established,related accept
        ct state invalid drop
        iif "lo" accept
        ip protocol icmp accept
        ip6 nexthdr icmpv6 accept
        ip saddr <IP Proxy Principal> tcp dport <Futuro puerto SSH> accept # proxy principal
        ip saddr <IP Proxy de segundo pais> tcp dport <Futuro puerto SSH> accept # proxy de respaldo 1
        ip saddr <IP Proxy de tercer pais> tcp dport <Futuro puerto SSH> accept # proxy de respaldo 2

        # Ya NO esta el "tcp dport 22 accept" de la parte 3. Nunca sirvio para
        # nada: el trafico del senuelo se resuelve en forward tras el DNAT, y
        # en el 22 del anfitrion no escucha ningun servicio, porque el SSH de
        # administracion esta en el puerto elegido. Dejarlo solo hacia creer que el
        # senuelo se abria desde aqui.
    }

    chain forward {
        type filter hook forward priority 0; policy drop;

        ct state established,related accept
        ct state invalid drop

        # --- Nada de un senuelo al otro -------------------------------------
        # Va PRIMERO, antes del ingress. Si Cowrie cae, no debe poder tocar el
        # 445 de Dionaea, y las reglas de ingress de abajo lo permitirian sin
        # esta linea, porque 172.20.0.20 es un destino valido en esos puertos.
        iifname "dirtyland0" oifname "dirtyland0" log prefix "DL-LATERAL " drop

        # --- INGRESS: AQUI SE EXPONEN LOS SENUELOS --------------------------
        # El "iifname != dirtyland0" es redundante con la regla anterior, y esa
        # redundancia es intencionada: si alguien reordena las reglas, la
        # exposicion no se convierte sola en un camino entre contenedores.
        iifname != "dirtyland0" oifname "dirtyland0" ip daddr 172.20.0.10 tcp dport { 2222, 2223 } accept
        iifname != "dirtyland0" oifname "dirtyland0" ip daddr 172.20.0.20 tcp dport { 135, 445, 1433 } accept

        # No se limita el ritmo de entrada a proposito: el senuelo existe para
        # ver todo lo que llega, y recortarlo seria tirar justo los datos que
        # queremos. El crecimiento del disco lo vigila el monitor del
        # anfitrion.

        # --- EGRESS de los contenedores (salida controlada) -----------------
        # Anti pivoteo: redes privadas, enlace local, CGNAT y las DOS IP
        # publicas del anfitrion.
        iifname "dirtyland0" ip saddr 172.20.0.0/24 ip daddr 10.0.0.0/8 log prefix "DL-EGRESS-PRIV " drop
        iifname "dirtyland0" ip saddr 172.20.0.0/24 ip daddr 172.16.0.0/12 log prefix "DL-EGRESS-PRIV " drop
        iifname "dirtyland0" ip saddr 172.20.0.0/24 ip daddr 192.168.0.0/16 log prefix "DL-EGRESS-PRIV " drop
        iifname "dirtyland0" ip saddr 172.20.0.0/24 ip daddr 169.254.0.0/16 log prefix "DL-EGRESS-PRIV " drop
        iifname "dirtyland0" ip saddr 172.20.0.0/24 ip daddr 100.64.0.0/10 log prefix "DL-EGRESS-PRIV " drop
        iifname "dirtyland0" ip saddr 172.20.0.0/24 ip daddr [IP VPS] log prefix "DL-EGRESS-HOST " drop
        iifname "dirtyland0" ip saddr 172.20.0.0/24 ip daddr [IP VPS 2] log prefix "DL-EGRESS-HOST " drop

        # Anti spam: SMTP saliente
        iifname "dirtyland0" ip saddr 172.20.0.0/24 tcp dport { 25, 465, 587 } log prefix "DL-EGRESS-SMTP " drop

        # Anti amplificacion: todo UDP saliente
        iifname "dirtyland0" ip saddr 172.20.0.0/24 meta l4proto udp log prefix "DL-EGRESS-UDP " drop

        # Salida TCP con limite de ritmo (descarga si, escaneo/DDoS no)
        iifname "dirtyland0" ip saddr 172.20.0.0/24 tcp flags syn limit rate 10/minute burst 20 packets accept
        iifname "dirtyland0" ip saddr 172.20.0.0/24 tcp flags syn log prefix "DL-EGRESS-RATE " drop
        iifname "dirtyland0" ip saddr 172.20.0.0/24 ip protocol tcp accept

        # Cajon de seguridad
        iifname "dirtyland0" ip saddr 172.20.0.0/24 log prefix "DL-EGRESS-OTHER " drop
    }

    chain output {
        type filter hook output priority 0; policy accept;
    }
}
